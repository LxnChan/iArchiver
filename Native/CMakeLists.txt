cmake_minimum_required(VERSION 3.24)
project(iArchiverNative VERSION 0.1.0 LANGUAGES C CXX)
option(IA_SIDELOAD "Include reviewed sideload-only operations" OFF)
option(IA_SHARED "Build a shared library for host integration tests" OFF)
find_package(LibArchive REQUIRED)
find_package(ZLIB REQUIRED)
find_package(LibLZMA REQUIRED)
if(IA_SHARED)
  set(IA_LIBRARY_TYPE SHARED)
else()
  set(IA_LIBRARY_TYPE STATIC)
endif()
include(LZMASDK.cmake)
add_library(ArchiveCore ${IA_LIBRARY_TYPE} ArchiveCore.c SevenZipBridge.cpp ${IA_SDK_SOURCES})
set_target_properties(ArchiveCore PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES WINDOWS_EXPORT_ALL_SYMBOLS YES POSITION_INDEPENDENT_CODE YES)
target_include_directories(ArchiveCore PUBLIC include)
target_include_directories(ArchiveCore PRIVATE "${IA_SDK}")
target_compile_definitions(ArchiveCore PRIVATE IA_SIDELOAD=$<BOOL:${IA_SIDELOAD}>)
if(IA_SIDELOAD)
  include(UnRAR.cmake)
endif()
target_link_libraries(ArchiveCore PUBLIC LibArchive::LibArchive ZLIB::ZLIB LibLZMA::LibLZMA)
if(WIN32)
  target_compile_definitions(ArchiveCore PRIVATE _CRT_SECURE_NO_WARNINGS LIBARCHIVE_STATIC)
  target_link_libraries(ArchiveCore PUBLIC bcrypt oleaut32 ole32 advapi32 uuid)
else()
  set_source_files_properties(ArchiveCore.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
  find_library(IA_ICONV iconv)
  if(IA_ICONV)
    target_link_libraries(ArchiveCore PUBLIC ${IA_ICONV})
  endif()
endif()
install(TARGETS ArchiveCore ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
